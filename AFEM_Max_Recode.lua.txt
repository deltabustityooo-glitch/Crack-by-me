local function cleanEntry(raw)
    if type(raw) ~= "table" then return nil end
    local id = raw.id or raw.Id
    if type(id) == "string" and builtinById[id] then
        local original = builtinById[id]
        return {id=id, name=original.name, description=original.description,
            animationid=original.animationid, builtin=original.builtin}
    end
    id = tonumber(id)
    if not id or id < 1 or id > 9007199254740991 then return nil end
    local name = raw.name or raw.Name
    if type(name) ~= "string" or name == "" then name = "Emote " .. tostring(id) end
    local description = raw.description or raw.Description
    if type(description) ~= "string" then description = "" end
    return {
        id = id, name = name:sub(1, 120), description = description:sub(1, 600),
        animationid = animationUrl(raw.animationid or raw.AnimationId or INCLUDED_BODY[id]),
        creatorName = raw.creatorName or raw.CreatorName,
        isOffSale = raw.isOffSale == true or raw.priceStatus == "OffSale",
    }
end

end

local resolved = {}
local resolving = {}
local function rawHttp(url)
    local ok, body = pcall(function() return game:HttpGet(url) end)
    if ok and type(body) == "string" and body ~= "" then return body end
    local requester = request or http_request or (syn and syn.request)
    if type(requester) == "function" then
        local good, response = pcall(requester, {Url=url, Method="GET"})
        if good and type(response) == "table" and type(response.Body) == "string"
            and response.Success ~= false and (tonumber(response.StatusCode) or 200) < 400 then
            return response.Body
        end
    end
    return nil
end

local function animationFromAssetBytes(assetId, version)
    local url = "https://assetdelivery.roblox.com/v1/asset/?id=" .. tostring(assetId)
    if version then url = url .. "&version=" .. tostring(version) end
    local blob = rawHttp(url)
    if not blob or not blob:find("AnimationId", 1, true) then return nil end
    -- Works with both RBXM binary properties and the older XML asset version.
    local id = blob:match("AnimationId.-rbxassetid://(%d+)")
        or blob:match("AnimationId.-[?&]id=(%d+)")
    return id and ("rbxassetid://" .. id) or nil
end

local function extractAnimation(item)
    -- Older Roblox emotes were later re-uploaded as FACE animations under the
    -- SAME catalog asset. Current GetObjects(id) now returns only facial motion.
    -- Asset version 1 still contains the original full-body animation.
    local legacy = item.creatorName == "Roblox" or item.id < 1000000000000
    if legacy then
        local body = animationFromAssetBytes(item.id, 1)
        if body then return body end
    end
    local candidates = {}
    local ok, objects = pcall(function() return game:GetObjects("rbxassetid://" .. tostring(item.id)) end)
    if ok and type(objects) == "table" then
        for _, object in ipairs(objects) do
            local function consider(animation, isRoot)
                local url = animationUrl(animation.AnimationId)
                if not url then return end
                local name = animation.Name:lower()
                local score = (isRoot and 10 or 0)
                if name == item.name:lower() then score = score + 20 end
                if name:find("body", 1, true) or name:find("emote", 1, true) then score = score + 8 end
                if name:find("face", 1, true) or name:find("facial", 1, true) then score = score - 100 end
                table.insert(candidates, {url=url, score=score})
            end
            if object:IsA("Animation") then consider(object, true) end
            for _, child in ipairs(object:GetDescendants()) do
                if child:IsA("Animation") then consider(child, false) end
            end
            object:Destroy()
        end
        table.sort(candidates, function(a, b) return a.score > b.score end)
        if candidates[1] and candidates[1].score >= 0 then return candidates[1].url end
    end
    -- If current asset was facial-only, try its original body before falling back.
    if not legacy then
        local olderBody = animationFromAssetBytes(item.id, 1)
        if olderBody then return olderBody end
    end
    local direct = animationFromAssetBytes(item.id)
    if direct then return direct end
    return candidates[1] and candidates[1].url or nil
end

local function resolveAnimation(item)
    local direct = animationUrl(item.animationid)
    if direct then return direct end
    if type(item.id) ~= "number" then return nil end
    local id = item.id
    if resolved[id] then
        rememberAnimation(item, resolved[id])
        return resolved[id]
    end
    local oldCache = animationUrl(getSetting("UGCEmote_" .. tostring(id)))
    if oldCache then
        resolved[id] = oldCache
        rememberAnimation(item, oldCache)
        return oldCache
    end
    if resolving[id] then
        local answer = resolving[id].Event:Wait()
        if answer then item.animationid = answer end
        return answer
    end
    local signal = Instance.new("BindableEvent")
    resolving[id] = signal
    local ok, result = pcall(extractAnimation, item)
    local answer = ok and animationUrl(result) or nil
    if answer then
        resolved[id] = answer
        rememberAnimation(item, answer)
    end
    resolving[id] = nil
    signal:Fire(answer)
    task.defer(function() signal:Destroy() end)
    return answer
end

local activeEmoteId
local function playEmote(item, fromList)
    if not alive or not item then return end
    local character = currentPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then
        notify("Emotes", "Дождись появления персонажа и попробуй снова.")
        return
    end
    if activeEmoteId == item.id and currentTrack and currentTrack.IsPlaying then
        currentTrack:Stop(0.15)
        activeEmoteId = nil
        return
    end
    if not item.builtin and humanoid.RigType ~= Enum.HumanoidRigType.R15 then
        notify("Emotes", "Каталожным эмоциям нужен персонаж R15.")
        return
    end
    local animationId = resolveAnimation(item)
    if not alive then return end
    if not animationId and not item.builtin then
        notify("Emotes", "Не удалось открыть анимацию: она удалена или недоступна в этой игре.")
        return
    end
    character = currentPlayer.Character
    humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return end
    if currentTrack then pcall(function() currentTrack:Stop(0.15) end); currentTrack = nil end

    local animate = character:FindFirstChild("Animate")
    local emoteFunction = animate and animate:FindFirstChild("PlayEmote")
    local played = false
    if emoteFunction and emoteFunction:IsA("BindableFunction") then
        local value = item.builtin or Instance.new("Animation")
        if not item.builtin then value.AnimationId = animationId end
        local ok, success, track = pcall(function() return emoteFunction:Invoke(value) end)
        played = ok and success ~= false and success ~= nil
        if played and typeof(track) == "Instance" and track:IsA("AnimationTrack") then currentTrack = track end
        if played and typeof(success) == "Instance" and success:IsA("AnimationTrack") then currentTrack = success end
        if not item.builtin then value:Destroy() end
    end
    if not played and animationId then
        local ok, track = pcall(function()
            local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)
            local animation = Instance.new("Animation")
            animation.AnimationId = animationId
            local loaded = animator:LoadAnimation(animation)
            loaded.Priority = Enum.AnimationPriority.Action
            loaded:Play(0.15)
            animation:Destroy()
            return loaded
        end)
        if ok then currentTrack = track; played = true end
    end
    if not played then
        notify("Emotes", "Игра или права на анимацию не позволяют её воспроизвести.")
    else
        activeEmoteId = item.id
        if fromList and closeAfterPlay then showMenu(false, true) end
    end
end

    currentTrack = nil
    activeEmoteId = nil
end)

-- Preview is built only after opening detail; it never blocks the catalog or UI.
local viewport = Converted["_AvatarPreview"]
local previewCamera
local previewAngle = 0
local function aimPreviewCamera()
    if not previewCamera then return end
    local angle = previewAngle
    previewCamera.CFrame = CFrame.lookAt(Vector3.new(math.sin(angle) * 6, 2, -math.cos(angle) * 6), Vector3.new(0, 2, 0))

local function previewAnimation(id)
    previewPendingAnimation = id
    if not previewModel or not id or not detail.Visible then return end
    local humanoid = previewModel:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    if previewTrack then pcall(function() previewTrack:Stop(0.12) end) end
    local ok, track = pcall(function()
        local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)
        local animation = Instance.new("Animation")
        animation.AnimationId = id
        local loaded = animator:LoadAnimation(animation)
        loaded.Priority = Enum.AnimationPriority.Action
        loaded:Play(0.12)
        animation:Destroy()
        return loaded
    end)
    previewTrack = ok and track or nil
end

local function buildPreview()
    previewBuildRevision = previewBuildRevision + 1
    local revision = previewBuildRevision
    if previewTrack then pcall(function() previewTrack:Stop(0) end); previewTrack = nil end
    if previewModel then previewModel:Destroy(); previewModel = nil end
    task.spawn(function()
        local description
        if previewOwn then
            local character = currentPlayer.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                local ok, applied = pcall(function() return humanoid:GetAppliedDescription() end)
                if ok then description = applied end
            end
        end
        if not description then
            description = Instance.new("HumanoidDescription")
            description.Head = 0
            description.Face = 0
        end
        local ok, model = pcall(function()
            return Players:CreateHumanoidModelFromDescriptionAsync(description, Enum.HumanoidRigType.R15)
        end)
        description:Destroy()
        if not ok or not model then return end
        if not alive or revision ~= previewBuildRevision then model:Destroy(); return end
        for _, descendant in ipairs(model:GetDescendants()) do
            if descendant:IsA("Script") or descendant:IsA("LocalScript") then
                descendant:Destroy()
            elseif not previewOwn and descendant:IsA("BasePart") then
                descendant.Color = Color3.fromRGB(163, 162, 165)
            end
        end
        if not previewOwn then
            Instance.new("Shirt", model).ShirtTemplate = "rbxassetid://18332758211"
            Instance.new("Pants", model).PantsTemplate = "rbxassetid://6087965851"
        end
        model:PivotTo(CFrame.new())
        model.Parent = viewport.WorldModel
        previewModel = model
        if not previewCamera then
            previewCamera = Instance.new("Camera")
            previewCamera.Parent = viewport
            viewport.CurrentCamera = previewCamera
        end
        aimPreviewCamera()
        if previewPendingAnimation then previewAnimation(previewPendingAnimation) end
    end)
end

local cardData = {}
local function makeCard(style)
    local template = style == "Wide" and Converted["_ItemListTemplate"] or Converted["_ItemListTemplateSquare"]
    local card = template:Clone()
    card.Name = "EmoteCard"
    card.Visible = true
    local click = card.Clickable
    local holding, holdSerial, usedLongHold = false, 0, false
    local infoDownAt = 0
    click.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton2 then
            if cardData[card] then showDetail(cardData[card], click) end
        elseif input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            holding = true
            usedLongHold = false
            holdSerial = holdSerial + 1
            local number = holdSerial
            task.delay(0.55, function()
                if alive and holding and holdSerial == number and card.Parent and cardData[card] then
                    usedLongHold = true
                    showDetail(cardData[card], click)
                end
            end)
        end
    end)
    click.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            holding = false
        end
    end)
    click.Activated:Connect(function()
        if not card.Parent or not cardData[card] or usedLongHold then return end
        if os.clock() - infoDownAt < 0.22 then return end
        task.spawn(playEmote, cardData[card], true)
    end)
    if style == "Wide" then
        local info = click.Details.Buttons.Play
        info.Text = "INFO"
        info.Size = UDim2.fromOffset(45, 20)
        info.InputBegan:Connect(function() infoDownAt = os.clock() end)
        info.Activated:Connect(function()
            infoDownAt = os.clock()
            if cardData[card] then showDetail(cardData[card], click) end
        end)
    end
    return card
end

local function renderCards(items)
    for _, pool in pairs(cardPools) do
        for _, card in ipairs(pool) do
            cardData[card] = nil
            card.Visible = false
            card.Parent = nil
        end
    end
    local pool = cardPools[format]
    for index, item in ipairs(items) do
        if index > 21 then break end
        local card = pool[index]
        if not card then
            card = makeCard(format)
            pool[index] = card
        end
        cardData[card] = item
        card.LayoutOrder = index

        click.UIStroke.CustomEffect.Enabled = (item.builtin ~= nil or (item.creatorName ~= nil and item.creatorName ~= "Roblox"))
        if format == "Wide" then
            click.Details.Title.Text = item.name
            click.Details.Description.Text = item.description or ""
            click.Details.Buttons.OffSale.Visible = item.isOffSale == true
            card.Size = UDim2.fromOffset(200, 100)
        else
            click.TitleHolder.TextLabel.Text = item.name
            card.Size = UDim2.fromOffset(100, 100)
        end
        card.Parent = listing
        card.Visible = true
    end
end

    if not item then return end

    if index then

        if button then button:Destroy(); floatButtons[item.id] = nil end
        table.remove(pins, index)
        notify("Floating B", "Кнопка убрана: " .. item.name)
    else
        if #pins >= 16 then
            notify("Floating B", "Можно закрепить не более 16 эмоций.")
            return
        end
        local pin = cleanEntry(item)
        if not pin then return end
        table.insert(pins, pin)

        remove.Activated:Connect(function()
            floatSuppressed[button] = true

    end
    writeList("floatingEmoteEntries", pins)

    syncFloats()
end

local PER_PAGE = 21
local states = {}
local activeTab = "Roblox"
local activeQuery = ""
local activePage = 1
local function stateKey(tab, query) return tab .. "\0" .. query end
local function stateFor(tab, query)
    local key = stateKey(tab, query)
    local state = states[key]
    if not state then
        state = {
            tab=tab, query=query, pages={}, buffer={}, seen={},
            cursor=nil, finished=false, loading=false, backend="http", apiCalls=0,
        }
        states[key] = state
    end
    return state
end

local function normalizeCatalog(raw, tab)
    if type(raw) ~= "table" then return nil end
    local kind = raw.assetType or raw.AssetType
    if kind and tonumber(kind) ~= 61 and tostring(kind) ~= "EmoteAnimation" then return nil end
    local itemType = raw.itemType or raw.ItemType
    if itemType and tostring(itemType) ~= "Asset" then return nil end
    local creatorId = tonumber(raw.creatorTargetId or raw.CreatorTargetId or raw.CreatorId)
    if tab == "Roblox" and creatorId and creatorId ~= 1 then return nil end
    local normalized = cleanEntry({
        id=raw.id or raw.Id, name=raw.name or raw.Name,
        description=raw.description or raw.Description,
        creatorName=raw.creatorName or raw.CreatorName,
        priceStatus=raw.priceStatus or raw.PriceStatus,
    })
    return normalized
end

local function searchUrl(state)
    local url = "https://catalog.roblox.com/v2/search/items/details?AssetTypeIds=61&IncludeNotForSale=true&Limit=30"
    if state.tab == "Roblox" then
        url = url .. "&CreatorType=1&CreatorTargetId=1"
    elseif state.query == "" then
        url = url .. "&SortType=3"
    end
    if state.query ~= "" then url = url .. "&Keyword=" .. HttpService:UrlEncode(state.query) end
    if state.cursor then url = url .. "&Cursor=" .. HttpService:UrlEncode(state.cursor) end
    return url
end

local function searchJson(url)
    for attempt = 1, 2 do
        local body = rawHttp(url)
        if body then
            local ok, response = pcall(function() return HttpService:JSONDecode(body) end)
            if ok and type(response) == "table" and type(response.data) == "table" then
                return response
            end
        end
        if attempt == 1 then task.wait(0.35) end
    end
    return nil, "Не удалось получить каталог Roblox (HTTP / сеть)."
end

local function startNative(state)
    local ok, pages = pcall(function()
        local params = CatalogSearchParams.new()
        params.AssetTypes = {Enum.AvatarAssetType.EmoteAnimation}
        params.IncludeOffSale = true
        params.Limit = 30
        if state.tab == "Roblox" then
            params.CreatorType = Enum.CreatorTypeFilter.User
            params.CreatorId = 1
        elseif state.query == "" then
            params.SortType = Enum.CatalogSortType.RecentlyCreated
        end
        if state.query ~= "" then params.SearchKeyword = state.query end
        return AvatarEditorService:SearchCatalogAsync(params)
    end)
    if not ok or not pages then return nil, "Каталог недоступен (HTTP и AvatarEditorService)." end
    state.backend = "native"
    state.native = pages
    state.nativeFirst = true
    return true
end

local function loadCatalogChunk(state)
    if state.backend == "native" then
        if state.nativeFirst then
            state.nativeFirst = false
        else
            if state.native.IsFinished then
                state.finished = true
                return {}
            end
            local ok = pcall(function() state.native:AdvanceToNextPageAsync() end)
            if not ok then return nil, "Не удалось открыть следующую страницу каталога." end
        end
        local ok, data = pcall(function() return state.native:GetCurrentPage() end)
        if not ok or type(data) ~= "table" then return nil, "Не удалось прочитать страницу каталога." end
        state.finished = state.native.IsFinished
        return data
    end
    local response, err = searchJson(searchUrl(state))
    if not response then
        if state.apiCalls == 0 then
            local ok, nativeError = startNative(state)
            if not ok then return nil, nativeError or err end
            return loadCatalogChunk(state)
        end
        return nil, err
    end
    local previousCursor = state.cursor
    state.cursor = response.nextPageCursor
    if state.cursor == "" or state.cursor == previousCursor then state.cursor = nil end
    state.finished = state.cursor == nil
    state.apiCalls = state.apiCalls + 1
    return response.data
end

local function assemblePage(state, number)
    local items = {}
    if number == 1 and state.tab == "Roblox" then
        for _, builtin in ipairs(BUILTIN) do
            if state.query == "" or builtin.name:lower():find(state.query:lower(), 1, true)
                or builtin.builtin:find(state.query:lower(), 1, true) then
                table.insert(items, builtin)
            end
        end
    end
    local emptyChunks = 0
    while #items + #state.buffer < PER_PAGE and not state.finished do
        local data, err = loadCatalogChunk(state)
        if not data then return nil, err end
        local before = #state.buffer
        for _, raw in ipairs(data) do
            local item = normalizeCatalog(raw, state.tab)
            if item and not state.seen[item.id] then
                state.seen[item.id] = true
                table.insert(state.buffer, item)
            end
        end
        if #state.buffer == before then emptyChunks = emptyChunks + 1 else emptyChunks = 0 end
        if emptyChunks >= 3 then break end -- Do not spin on hidden/empty API pages.
    end
    local remaining = PER_PAGE - #items
    for _ = 1, math.min(remaining, #state.buffer) do
        table.insert(items, table.remove(state.buffer, 1))
    end
    if #items == 0 and not state.finished then
        return nil, "Пустой ответ каталога: попробуй снова или измени поиск."
    end
    return {items=items, hasMore=(#state.buffer > 0 or not state.finished)}
end

local function fetchPage(state, number)
    if state.pages[number] then return state.pages[number] end
    if number ~= #state.pages + 1 then return nil, "Открой страницы по порядку." end
    if state.loading then
        while alive and state.loading do task.wait(0.04) end
        return state.pages[number], state.error
    end
    state.loading = true
    state.error = nil
    local ok, page, err = pcall(assemblePage, state, number)
    state.loading = false
    if not ok then
        state.error = tostring(page)
    elseif page then
        state.pages[number] = page
    else
        state.error = err
    end
    return state.pages[number], state.error
end

local function setBusy(busy)
    actionBar.Spinner.Visible = busy
    actionBar.Spinner.Size = UDim2.fromOffset(24, 24)
    if busy and not spinTween then
        spinTween = TweenService:Create(actionBar.Spinner.Spinner,
            TweenInfo.new(0.8, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1), {Rotation=360})
        spinTween:Play()
    elseif not busy and spinTween then
        spinTween:Cancel()
        spinTween = nil
        actionBar.Spinner.Spinner.Rotation = 0
    end
end

local function fallbackItems()
    if activeTab ~= "Roblox" or activePage ~= 1 then return {} end
    local output = {}
    for _, source in ipairs({BUILTIN, OFFLINE_ROBLOX}) do
        for _, item in ipairs(source) do
            if activeQuery == "" or item.name:lower():find(activeQuery:lower(), 1, true) then
                table.insert(output, item)
            end
        end
    end
    return output
end

local function updatePager(state, page)
    for _, child in ipairs(pager:GetChildren()) do
        if child:GetAttribute("GeneratedPage") then child:Destroy() end
    end
    local last = #state.pages
    if page and page.hasMore then last = last + 1 end
    if not page and activePage == 1 then last = 1 end
    local firstVisible = math.max(1, activePage - 2)
    for index = firstVisible, math.min(last, activePage + 2) do
        local dot = pager.SamplePage:Clone()
        dot.Name = "Page" .. tostring(index)
        dot:SetAttribute("GeneratedPage", true)
        dot.Label.Text = tostring(index)
        dot.Label.TextTransparency = index == activePage and 0 or 0.7
        dot.UIScale.Scale = index == activePage and 1 or 0.55
        dot.LayoutOrder = index
        dot.Visible = true
        dot.Parent = pager
    end
    local hasNext = page and page.hasMore == true or false
    pager.Next.UIScale.Scale = hasNext and 1 or 0.0001
    pager.Next.Interactable = hasNext
    pager.Previous.UIScale.Scale = activePage > 1 and 1 or 0.0001
    pager.Previous.Interactable = activePage > 1
end

local function drawActive()
    local state = stateFor(activeTab, activeQuery)
    local page = state.pages[activePage]
    renderCards(page and page.items or fallbackItems())
    updatePager(state, page)
end

local function loadActive()
    catalogRevision = catalogRevision + 1
    local revision = catalogRevision
    local state = stateFor(activeTab, activeQuery)
    local requestedPage = activePage
    if state.pages[requestedPage] then
        setBusy(false)
        drawActive()
        listing.CanvasPosition = Vector2.new(0, 0)
        return
    end
    drawActive()
    setBusy(true)
    task.spawn(function()
        local result, err = fetchPage(state, requestedPage)
        if not alive or revision ~= catalogRevision then return end
        setBusy(false)
        if result then
            drawActive()
            listing.CanvasPosition = Vector2.new(0, 0)
        else
            if requestedPage > 1 then activePage = requestedPage - 1 end
            drawActive()
            notify("Emotes", err or "Каталог сейчас недоступен. Попробуй ещё раз.")
        end
    end)
end

local function switchTab(tab)
    if tab ~= "Roblox" and tab ~= "UGC" then return end
    activeTab = tab
    activePage = 1
    activeQuery = actionBar.Search.Text:match("^%s*(.-)%s*$"):sub(1, 100)
    local switch = actionBar.SwitchTabs
    for _, button in ipairs({switch.Roblox, switch.UGC}) do
        local selected = button.Name == tab
        tween(button, {
            BackgroundTransparency = selected and 0 or 1,
            TextColor3 = selected and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 255, 255),
        }, 0.12)
    end
    -- Original TrendingUGC is only a skeleton; do not reserve empty space.
    emotesUI.TrendingUGC.Visible = false
    loadActive()
end
